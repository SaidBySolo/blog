## 코드 블록

```ts
// TypeScript 예시
interface User {
  id: number;
  name: string;
}

function greet(user: User): string {
  return `Hello, ${user.name}!`;
}

console.log(greet({ id: 1, name: "Solo" }));
```

```python
# Python 예시
def fibonacci(n: int) -> list[int]:
    result = [0, 1]
    for _ in range(n - 2):
        result.append(result[-1] + result[-2])
    return result[:n]

print(fibonacci(10))
```
